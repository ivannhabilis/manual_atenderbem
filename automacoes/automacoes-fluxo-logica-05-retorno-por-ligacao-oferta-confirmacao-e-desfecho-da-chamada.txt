confirma_ligacao: question varName="quer_ligacao" text="Responda SIM para receber a ligação ou NAO para continuar por mensagem."
decide_retorno: varCondition value="{{quer_ligacao | toUpperCase}}" {
  case aceita conditionOperator="in" conditionValue="SIM" {
    confirma_retorno: message text="Combinado! Registrei o pedido de ligação para o número {{chat_client_number}}."
    registra_pedido: info text="Cliente aceitou o retorno por telefone. O pedido fica registrado para a discagem pela automação de voz."
    pausa: sleep time=2
    aguarda_cliente: waitMessage
    apos_ligacao: message text="Se a ligação não completou, escreva aqui que eu sigo com você por mensagem."
    fim_retorno: terminate reason="Retorno por telefone solicitado"
  }
  else {
    continua_chat: message text="Sem problema! Seguimos por aqui. Escreva o que você precisa que eu te ajudo."
    fim_chat: terminate reason="Cliente preferiu continuar por mensagem"
  }
}
