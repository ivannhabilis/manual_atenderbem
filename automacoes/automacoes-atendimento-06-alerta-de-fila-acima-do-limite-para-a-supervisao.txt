le_fila: queueChats status=1 resultVariableName="aguardando" queueId="3" limit=100
checa: varCondition {
  case acima conditionOperator="expression" conditionValue="aguardando_count * 1 > 5" {
    alerta: sendAlert usersIds=["supervisor_plantao"] text="Fila de suporte acima do limite: {{aguardando_count}} atendimento(s) aguardando agente."
    notifica: sendNotification selectionMode=0 userId="supervisor_plantao" text="Verifique a fila de suporte: {{aguardando_count}} aguardando atendente." type=2
    fim_alerta: terminate reason="Fila acima do limite" obs="Alerta enviado a supervisao."
  }
  else {
    tranquilo: sendNotification selectionMode=0 userId="supervisor_plantao" text="Fila de suporte sob controle: {{aguardando_count}} aguardando atendente." type=0
    fim_ok: terminate reason="Fila sob controle" obs="Nenhum alerta necessario."
  }
}
