comeca_resumo: newVar variableName="resumo_carrinho" value=""
confere_carrinho: varCondition value="{{chat_cart_items_count}}" {
  case carrinho_vazio conditionOperator="<" conditionValue="1" {
    sem_itens: message text="Seu carrinho está vazio no momento, {{chat_client_name}}. Quer ver as novidades da loja?"
    fim_sem_itens: terminate reason="Carrinho vazio"
  }
  else {
    cabecalho: message text="Confira o que já está no seu carrinho:"
    prepara_titulo: jscode code="vars[\"resumo_carrinho\"] = \"Carrinho com \" + vars[\"chat_cart_items_count\"] + \" itens\""
    laco_itens: loop mode="forEach" list="{{chat_cart_lines}}" itemVar="item" indexVar="posicao" {
      body {
        mostra_item: message text="{{posicao}}. {{item.product_name}} - quantidade {{item.product_quantity}} - valor unitário {{item.product_value}}"
        pausa_envio: sleep time=1 -> laco_itens
      }
    }
    mostra_total: message text="{{resumo_carrinho}}. Total: {{chat_cart_total}}."
    fim_com_itens: terminate reason="Resumo do carrinho enviado"
  }
}
